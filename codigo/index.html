lib/ai/support.ts

/**
 * Detección de soporte y disponibilidad para cada API built-in.
 *
 * Regla de oro de la charla: NUNCA asumas que el modelo está listo.
 *   1. Feature detection  → ¿existe la API en este navegador?   ('X' in self)
 *   2. availability()     → ¿este equipo puede usarla ya?
 *                           'available' | 'downloadable' | 'downloading' | 'unavailable'
 *   3. create()           → dentro de un gesto del usuario (clic, tecla) si hay que descargar.
 *
 * Los globals de IA built-in (LanguageModel, Translator…) solo existen en el
 * navegador. Llama a estos helpers desde efectos o handlers, nunca durante el
 * render ni en SSR.
 */

export type ApiName = 'LanguageModel' | 'Summarizer' | 'Rewriter' | 'Translator' | 'LanguageDetector';

/** Los cuatro estados oficiales de availability() + uno propio: el navegador ni siquiera tiene la API. */
export type ApiStatus = Availability | 'no-soportada';

/** Idioma usado en toda la charla (español). availability() depende del idioma que declares. */
export const ES = ['es'];

/** Paso 1 · Feature detection segura para SSR (`typeof self` evita el ReferenceError en el servidor). */
export function hasApi(name: ApiName): boolean {
  return typeof self !== 'undefined' && name in self;
}

/**
 * Paso 2 · availability() con LAS MISMAS opciones que usará create().
 * La respuesta cambia según idioma y modalidad: preguntar sin opciones puede
 * decir 'available' y luego create() fallar con el idioma real.
 */
export async function checkApi(name: ApiName): Promise<ApiStatus> {
  if (!hasApi(name)) return 'no-soportada';
  try {
    switch (name) {
      case 'LanguageModel':
        return await LanguageModel.availability({
          expectedInputs: [{ type: 'text', languages: ES }],
          expectedOutputs: [{ type: 'text', languages: ES }],
        });
      case 'Summarizer':
        return await Summarizer.availability({ expectedInputLanguages: ['es', 'en'], outputLanguage: 'es' });
      case 'Rewriter':
        return await Rewriter.availability({ expectedInputLanguages: ES, outputLanguage: 'es' });
      case 'Translator':
        // Solo consulta el par en→es (el del gancho). Cada par de idiomas tiene su propio paquete.
        return await Translator.availability({ sourceLanguage: 'en', targetLanguage: 'es' });
      case 'LanguageDetector':
        return await LanguageDetector.availability();
    }
  } catch {
    // Opciones no soportadas o bloqueo por política: se muestra como 'unavailable', la tabla no se rompe.
    return 'unavailable';
  }
}

/** Orden de las filas en la tabla de la Demo 01. */
export const API_NAMES: ApiName[] = ['LanguageModel', 'LanguageDetector', 'Translator', 'Summarizer', 'Rewriter'];

/**
 * Monitor reutilizable que reporta el progreso de descarga (0 → 1).
 * Se pasa como `monitor` a cualquier create(). Es lo que alimenta la barra de
 * la Demo 01: "el usuario ve qué está pasando, no una pantalla congelada".
 */
export function progressMonitor(onProgress: (fraction: number) => void): CreateMonitorCallback {
  return (monitor) => {
    monitor.addEventListener('downloadprogress', (event) => {
      // En la spec actual `loaded` ya es una fracción de 0 a 1 (no bytes): no hay que dividir por `total`.
      // Si el modelo ya está descargado, este evento puede no llegar nunca.
      onProgress((event as ProgressEvent).loaded);
    });
  };
}

/**
 * Cachea una instancia por clave. Crear la instancia es lo caro (puede
 * descargar un modelo); usarla es barato. Por eso se reutiliza.
 *
 * - Se cachea la PROMESA: un doble clic espera la misma creación, no lanza dos.
 * - Una creación rechazada se expulsa para que el siguiente clic reintente
 *   (p. ej. el primer intento no tenía activación del usuario → NotAllowedError).
 */
export function memoize<K, V>(cache: Map<K, Promise<V>>, key: K, create: () => Promise<V>): Promise<V> {
  let entry = cache.get(key);
  if (!entry) {
    entry = create();
    cache.set(key, entry);
    const created = entry;
    // Solo expulsa SU entrada: un reset puede haberla reemplazado ya por una nueva.
    created.catch(() => {
      if (cache.get(key) === created) cache.delete(key);
    });
  }
  return entry;
}

/**
 * Destruye y expulsa todas las instancias cacheadas. Se usa tras un fallo en
 * tiempo de ejecución (p. ej. UnknownError) para que el siguiente intento se
 * cure solo, sin recargar la página. Clave para sobrevivir a una demo en vivo.
 */
export function clearCache<K, V extends { destroy(): void }>(cache: Map<K, Promise<V>>): void {
  for (const entry of cache.values()) {
    void entry.then((instance) => instance.destroy()).catch(() => {});
  }
  cache.clear();
}

export function errorMessage(error: unknown): string {
  return error instanceof Error ? error.message : String(error);
}

/** Cancelación intencional (el usuario siguió escribiendo). No es un error que mostrar. */
export function isAbortError(error: unknown): boolean {
  return (error as DOMException | undefined)?.name === 'AbortError';
}

/**
 * true para fallos del navegador (DOMException como UnknownError,
 * InvalidStateError o NotAllowedError) que pueden dejar rota una instancia
 * cacheada → hay que resetear el caché.
 * Nuestros errores de validación (Error normal) y los aborts NO cuentan: la
 * instancia sigue sana.
 */
export function isRuntimeFailure(error: unknown): boolean {
  return error instanceof DOMException && !isAbortError(error);
}

lib/ai/stream.ts

/**
 * Pinta un stream de texto sin bloquear la UI.
 *
 * Las APIs built-in no corren en Web Workers (Summarizer, por ejemplo), así que
 * en la página el trabajo pesado es renderizar. Pintar cada chunk dispararía un
 * render de React por token: acumulamos y pintamos como mucho UNA vez por frame
 * con requestAnimationFrame.
 *
 * Lo usan las Demos 02 (translate), 04 (summarize) y 05 (rewrite).
 */
export async function streamText(
  stream: ReadableStream<string>,
  onPaint: (text: string) => void,
  signal?: AbortSignal,
): Promise<string> {
  let buffer = '';
  let frame: number | undefined;
  // Al terminar (éxito o fallo) ningún frame tardío debe repintar: en un fallo
  // sobrescribiría el panel de error y lo devolvería a "streaming".
  let finished = false;

  const paint = () => {
    frame = undefined;
    if (!finished && !signal?.aborted) onPaint(buffer);
  };

  // Limpia el panel al empezar.
  if (!signal?.aborted) onPaint('');

  try {
    // ReadableStream es async-iterable en Chrome.
    for await (const chunk of stream) {
      if (signal?.aborted) break;
      // Los chunks son deltas (fragmentos nuevos), por eso se concatenan.
      buffer += chunk;
      // Como mucho un frame pendiente: 50 chunks en un frame = 1 render.
      frame ??= requestAnimationFrame(paint);
    }
    // Éxito: pinta el buffer final de forma síncrona.
    if (frame !== undefined) cancelAnimationFrame(frame);
    paint();
    return buffer;
  } finally {
    finished = true;
    if (frame !== undefined) cancelAnimationFrame(frame);
  }
}

lib/ai/translate.ts

/**
 * Demo 02 · Traducción en tiempo real.
 *
 * LanguageDetector identifica el idioma; Translator traduce con streaming.
 * Translator usa su propio modelo experto (NO Gemini Nano) con paquetes por
 * par de idiomas. Solo Chrome desktop.
 */
import { memoize, progressMonitor } from './support';

/** Callback opcional de progreso de descarga (fracción de 0 a 1). */
export type OnProgress = (fraction: number) => void;

let detector: Promise<LanguageDetector> | undefined;
const translators = new Map<string, Promise<Translator>>();

/**
 * Un Translator por par de idiomas: crearlo es lo caro (descarga el paquete
 * la primera vez). Clave del caché: 'en→es', 'es→pt'…
 * Ojo: `onProgress` solo se engancha en la creación; las llamadas siguientes reutilizan la instancia.
 */
export function getTranslator(sourceLanguage: string, targetLanguage: string, onProgress?: OnProgress) {
  return memoize(translators, `${sourceLanguage}→${targetLanguage}`, () =>
    Translator.create({
      sourceLanguage,
      targetLanguage,
      monitor: progressMonitor((f) => {
        console.info(`Translator ${sourceLanguage}→${targetLanguage}: ${Math.round(f * 100)}%`);
        onProgress?.(f);
      }),
    }),
  );
}

/** Detecta el idioma del texto. Devuelve el más probable y su confianza en %. */
export async function detectLanguage(text: string, signal?: AbortSignal, onProgress?: OnProgress) {
  if (!detector) {
    const created = LanguageDetector.create({
      monitor: progressMonitor((f) => {
        console.info(`LanguageDetector: ${Math.round(f * 100)}%`);
        onProgress?.(f);
      }),
    });
    detector = created;
    // Solo limpia su propia promesa: un reset puede haberla reemplazado ya.
    created.catch(() => {
      if (detector === created) detector = undefined;
    });
  }
  // detect() devuelve candidatos ordenados por confianza; nos quedamos con el primero.
  const [best] = await (await detector).detect(text, { signal });
  return {
    // Sin resultado (texto muy corto) asumimos inglés.
    language: best?.detectedLanguage ?? 'en',
    confidence: Math.round((best?.confidence ?? 0) * 100),
  };
}

/**
 * Descarta una instancia cacheada tras un fallo en tiempo de ejecución (p. ej.
 * UnknownError) para que el siguiente intento cree una nueva en vez de
 * reutilizar una rota. Los fallos de creación ya los expulsa memoize / el catch del detector.
 */
export function resetDetector() {
  void detector?.then((d) => d.destroy()).catch(() => {});
  detector = undefined;
}

export function resetTranslator(sourceLanguage: string, targetLanguage: string) {
  const key = `${sourceLanguage}→${targetLanguage}`;
  void translators.get(key)?.then((t) => t.destroy()).catch(() => {});
  translators.delete(key);
}

/** availability() por par → Translator cacheado → stream de texto traducido. */
export async function translateStreaming(
  text: string,
  source: string,
  target: string,
  signal?: AbortSignal,
  onProgress?: OnProgress,
) {
  // Regla de oro, paso 2: cada par tiene su propio estado ('downloadable' si falta el paquete).
  const availability = await Translator.availability({ sourceLanguage: source, targetLanguage: target });
  if (availability === 'unavailable') {
    throw new Error(`Par ${source} → ${target} no soportado.`);
  }
  const translator = await getTranslator(source, target, onProgress);
  // El `signal` permite cancelar la traducción si el usuario sigue escribiendo.
  return translator.translateStreaming(text, { signal });
}

lib/ai/structured.ts

/**
 * Demo 03 · Salida estructurada con JSON Schema.
 *
 * El schema va en prompt() como `responseConstraint`, NO en create(). El
 * navegador restringe la generación para que el modelo SOLO pueda producir JSON
 * válido según el schema: nada de "Claro, aquí tienes el JSON: ```json…".
 */
import { ES, isAbortError } from './support';

/** Tipo TypeScript que refleja el schema de abajo: lo que la UI recibe ya tipado. */
export interface Review {
  producto: string;
  sentimiento: 'positivo' | 'neutral' | 'negativo';
  puntaje: number;
  temas: string[];
  resumen: string;
}

/** JSON Schema estándar: `enum`, `minimum`/`maximum`, `maxItems` y `required` se respetan al generar. */
export const reviewSchema = {
  type: 'object',
  properties: {
    producto: { type: 'string' },
    sentimiento: { type: 'string', enum: ['positivo', 'neutral', 'negativo'] },
    puntaje: { type: 'number', minimum: 1, maximum: 5 },
    temas: { type: 'array', items: { type: 'string' }, maxItems: 4 },
    resumen: { type: 'string' },
  },
  required: ['producto', 'sentimiento', 'puntaje', 'temas', 'resumen'],
  additionalProperties: false,
} as const;

/** Sesión base, creada una sola vez, con el prompt de sistema ya cargado. */
let session: Promise<LanguageModel> | undefined;

function getSession() {
  session ??= LanguageModel.create({
    // Declarar idioma de entrada/salida: availability() y la calidad dependen de ello.
    expectedInputs: [{ type: 'text', languages: ES }],
    expectedOutputs: [{ type: 'text', languages: ES }],
    initialPrompts: [
      {
        role: 'system',
        content: 'Analizas reseñas de productos y extraes datos. Responde siempre en español.',
      },
    ],
  });
  // Si la creación falla (p. ej. falta activación), el siguiente clic reintenta.
  session.catch(() => (session = undefined));
  return session;
}

export async function analyzeReview(review: string): Promise<Review> {
  const sessionPromise = getSession();
  const s = await sessionPromise;

  // Cada prompt() se suma al contexto de la sesión; clonamos para que los
  // turnos no se acumulen (cada reseña se analiza "desde cero" con el mismo prompt de sistema).
  let turn: LanguageModel;
  try {
    turn = await s.clone();
  } catch (error) {
    // La sesión base está rota: se expulsa para que el siguiente intento cree otra.
    s.destroy();
    if (session === sessionPromise) session = undefined;
    throw error;
  }
  try {
    let result: string;
    try {
      // ⭐ La línea clave de la demo: el schema como responseConstraint.
      result = await turn.prompt(`Analiza esta reseña:\n\n${review}`, {
        responseConstraint: reviewSchema,
      });
    } catch (error) {
      // Un fallo en tiempo de ejecución puede dejar rota también la sesión base: se expulsa.
      // Los errores de JSON.parse de abajo son problemas de la salida, así que conservan la sesión.
      if (!isAbortError(error) && session === sessionPromise) {
        s.destroy();
        session = undefined;
      }
      throw error;
    }
    // La salida es un string JSON garantizado por el schema: JSON.parse directo.
    return JSON.parse(result) as Review;
  } finally {
    // El clon se destruye siempre para liberar memoria del modelo.
    turn.destroy();
  }
}

/**
 * Demo 04 · Resúmenes locales.
 *
 * Summarizer no funciona en Web Workers: todo vive en el hilo principal.
 * Defensas:
 *   - streaming + pintado con requestAnimationFrame (ver lib/ai/stream.ts)
 *   - medir la entrada contra inputQuota ANTES de enviarla.
 */
import { clearCache, memoize, progressMonitor } from './support';

/** Un Summarizer por combinación tipo×longitud (`'key-points:short'`…): las opciones se fijan en create(). */
const cache = new Map<string, Promise<Summarizer>>();

/** Descarta todos los summarizers cacheados tras un fallo en tiempo de ejecución. */
export function resetSummarizers() {
  clearCache(cache);
}

export function getSummarizer(type: SummarizerType, length: SummarizerLength) {
  return memoize(cache, `${type}:${length}`, () =>
    Summarizer.create({
      type, // 'key-points' | 'tldr' | 'teaser' | 'headline'
      length, // 'short' | 'medium' | 'long'
      format: 'plain-text', // el panel es un <pre>: sin markdown
      expectedInputLanguages: ['es', 'en'],
      outputLanguage: 'es',
      // Contexto fijo que orienta todos los resúmenes de esta instancia.
      sharedContext: 'Artículos técnicos para desarrolladores web',
      monitor: progressMonitor((f) => console.info(`Summarizer: ${Math.round(f * 100)}%`)),
    }),
  );
}

/**
 * Antes de enviar: ¿cabe el texto? Si no, trocéalo (resumen de resúmenes).
 * measureInputUsage() cuenta tokens con el tokenizador real del modelo.
 */
export async function checkQuota(summarizer: Summarizer, text: string) {
  const usage = await summarizer.measureInputUsage(text);
  return { usage, quota: summarizer.inputQuota, fits: usage <= summarizer.inputQuota };
}

lib/ai/hybrid.ts

/**
 * Demo 06 · Puente híbrido de supervivencia.
 *
 * "Local cuando se puede, nube cuando no." Dos estrategias:
 *  - manual:   tú decides el fallback ANTES de fallar (feature detection →
 *              availability → try/catch como última red de seguridad).
 *  - firebase: decide Firebase AI Logic (PREFER_ON_DEVICE).
 *
 * En escena: ?force=cloud en la URL simula `unavailable` sin tocar flags ni
 * reiniciar Chrome.
 */
import { ES, hasApi, isAbortError } from './support';
import { InferenceMode, getHybridModel } from './firebase';

/** Quién respondió. Los mismos valores que `response.inferenceSource` de Firebase. */
export type Source = 'on_device' | 'in_cloud';
export interface Result {
  text: string;
  source: Source;
}

/** true si la URL lleva ?force=cloud. SSR-safe. */
export const forceCloud = () =>
  typeof location !== 'undefined' && new URLSearchParams(location.search).get('force') === 'cloud';

const localOptions = {
  expectedInputs: [{ type: 'text' as const, languages: ES }],
  expectedOutputs: [{ type: 'text' as const, languages: ES }],
};

/**
 * Sesión base, creada una sola vez. create() es lo caro (~20 s en frío);
 * clone() es barato, así que cada llamada trabaja sobre un clon.
 */
let baseSession: Promise<LanguageModel> | undefined;

function getBaseSession(): Promise<LanguageModel> {
  baseSession ??= LanguageModel.create(localOptions);
  // Si la creación falla, el siguiente intento vuelve a crear.
  baseSession.catch(() => (baseSession = undefined));
  return baseSession;
}

/** Expulsa la sesión base (p. ej. tras un fallo) para que la próxima llamada cree otra. */
function dropBaseSession(): void {
  void baseSession?.then((s) => s.destroy()).catch(() => {});
  baseSession = undefined;
}

/** ¿Se puede usar el modelo local ahora mismo? NUNCA lanza. */
async function canUseLocal(allowLocal = true): Promise<boolean> {
  if (!allowLocal || forceCloud()) return false; // simulación en escena (o el juego LULO: "la nave no tiene modelo")
  if (!hasApi('LanguageModel')) return false; // paso 1: el navegador no tiene la API
  try {
    // Paso 2: solo 'available' sirve. 'downloadable' también va a la nube:
    // no hacemos esperar al usuario una descarga de varios GB para una respuesta.
    return (await LanguageModel.availability(localOptions)) === 'available';
  } catch {
    return false;
  }
}

/**
 * Precalienta la sesión base para que el primer "Generar" no pague el create().
 * Solo si el modelo ya está 'available' (nunca dispara una descarga). NUNCA lanza.
 */
export async function warmUpLocal(): Promise<void> {
  if (!(await canUseLocal())) return;
  await getBaseSession().catch(() => {});
}

/**
 * Devuelve un clon de la sesión local o null. NUNCA lanza.
 * Toda la decisión "¿puedo usar el modelo local?" vive aquí, antes de fallar.
 */
async function getLocalSession(allowLocal = true): Promise<LanguageModel | null> {
  if (!(await canUseLocal(allowLocal))) return null;
  try {
    // Clonamos para que los turnos no se acumulen en el contexto de la sesión base.
    return await (await getBaseSession()).clone();
  } catch {
    dropBaseSession(); // p. ej. falta memoria o la sesión base quedó rota
    return null;
  }
}

/** Ruta en la nube: Gemini vía Firebase AI Logic, forzando ONLY_IN_CLOUD. */
async function callCloud(prompt: string): Promise<string> {
  const model = getHybridModel(InferenceMode.ONLY_IN_CLOUD);
  const { response } = await model.generateContent(prompt);
  return response.text();
}

/** Estrategia manual: el patrón que puedes portar a cualquier backend (OpenAI, tu API, etc.). */
export async function generateManual(prompt: string, { allowLocal = true }: { allowLocal?: boolean } = {}): Promise<Result> {
  const session = await getLocalSession(allowLocal);
  if (session) {
    try {
      return { text: await session.prompt(prompt), source: 'on_device' };
    } catch (error) {
      // try/catch es la última red de seguridad: un fallo local en tiempo de ejecución cae a la nube.
      if (isAbortError(error)) throw error; // una cancelación no debe gastar nube
      console.warn('[híbrido] Falló la inferencia local', error);
      dropBaseSession(); // un fallo en ejecución puede dejar rota también la base
    } finally {
      // El clon es de un solo uso: se libera siempre. La base sigue viva.
      session.destroy();
    }
  }
  // Esta línea en la consola es la prueba en escena de que entró la nube.
  console.warn('[híbrido] Fallback a la nube');
  return { text: await callCloud(prompt), source: 'in_cloud' };
}

/** Estrategia Firebase: un solo código, dos runtimes. El SDK decide y te dice quién respondió. */
export async function generateFirebase(prompt: string): Promise<Result> {
  const mode = forceCloud() ? InferenceMode.ONLY_IN_CLOUD : InferenceMode.PREFER_ON_DEVICE;
  const { response } = await getHybridModel(mode).generateContent(prompt);
  return { text: response.text(), source: (response.inferenceSource as Source) ?? 'in_cloud' };
}

/**
 * Firebase AI Logic: inferencia híbrida.
 *
 * Usa Gemini Nano (Prompt API) cuando está disponible y cae a Gemini en la
 * nube cuando no. Requiere NEXT_PUBLIC_FIREBASE_* en .env.local
 * (consola de Firebase → AI Logic → Gemini Developer API).
 *
 * Sin configuración, la Demo 06 solo funciona por la ruta local y la nube
 * falla con un mensaje claro.
 * Docs: docs/demos/06-hybrid.md.
 */
import { initializeApp, type FirebaseApp } from 'firebase/app';
import { GoogleAIBackend, InferenceMode, getAI, getGenerativeModel, type GenerativeModel } from 'firebase/ai';
import { ReCaptchaEnterpriseProvider, initializeAppCheck } from 'firebase/app-check';

declare global {
  var FIREBASE_APPCHECK_DEBUG_TOKEN: boolean | string | undefined;
}

// Acceso literal a process.env para que Next.js inserte los valores en el bundle del cliente.
// (Son claves públicas de Firebase: la seguridad va en App Check y reglas, no en ocultarlas.)
const config = {
  apiKey: process.env.NEXT_PUBLIC_FIREBASE_API_KEY,
  authDomain: process.env.NEXT_PUBLIC_FIREBASE_AUTH_DOMAIN,
  projectId: process.env.NEXT_PUBLIC_FIREBASE_PROJECT_ID,
  appId: process.env.NEXT_PUBLIC_FIREBASE_APP_ID,
};

const recaptchaSiteKey = process.env.NEXT_PUBLIC_RECAPTCHA_SITE_KEY;

export const isFirebaseConfigured = Boolean(config.apiKey && config.projectId);

let app: FirebaseApp | undefined;

/**
 * App Check: el proxy de AI Logic rechaza (401) las requests sin token cuando el
 * proyecto tiene enforcement activo. En localhost reCAPTCHA no valida, así que en
 * desarrollo usamos un debug token registrado en la consola (App Check → Manage debug tokens).
 * Sin NEXT_PUBLIC_FIREBASE_APPCHECK_DEBUG_TOKEN, el SDK genera uno y lo imprime en la consola del navegador.
 */
function setupAppCheck(firebaseApp: FirebaseApp): void {
  if (typeof window === 'undefined') return;
  if (!recaptchaSiteKey) {
    console.warn('[firebase] App Check desactivado: falta NEXT_PUBLIC_RECAPTCHA_SITE_KEY.');
    return;
  }
  if (process.env.NODE_ENV === 'development') {
    // Dentro de este branch para que el token no llegue al bundle de producción.

"use client";

/**
 * Demo 02 · Traducción en tiempo real (LanguageDetector + Translator).
 */
import { useEffect, useRef, useState } from "react";
import { Output, useOutput } from "@/components/output";
import { streamText } from "@/lib/ai/stream";
import { hasApi, isAbortError, isRuntimeFailure } from "@/lib/ai/support";
import {
  detectLanguage,
  resetDetector,
  resetTranslator,
  translateStreaming,
} from "@/lib/ai/translate";

const INITIAL =
  "The best part of on-device AI is that your users' data never leaves their laptop. No API keys, no tokens, no invoices.";

const TARGETS = [
  { value: "es", label: "Español" },
  { value: "en", label: "English" },
  { value: "pt", label: "Português" },
  { value: "fr", label: "Français" },
];

export function TranslateDemo() {
  const [text, setText] = useState(INITIAL);
  const [target, setTarget] = useState("es");
  const [detected, setDetected] = useState("Idioma detectado: —");
  const output = useOutput();
  // Una ejecución en curso a la vez: cada nueva cancela la anterior.
  const controller = useRef<AbortController>();
  const debounceTimer = useRef<ReturnType<typeof setTimeout>>();

  // Al salir de la página: cancela el debounce y la traducción pendiente.
  useEffect(
    () => () => {
      clearTimeout(debounceTimer.current);
      controller.current?.abort();
    },
    [],
  );

  async function run(value: string, targetLanguage: string) {
    // Cancelar primero, para que una ejecución vieja nunca pinte sobre un input vacío.
    controller.current?.abort();
    const input = value.trim();
    if (!input) {
      controller.current = undefined;
      setDetected("Idioma detectado: —");
      output.show("", "idle");
      return;
    }

    controller.current = new AbortController();
    const { signal } = controller.current;

    // Arranque en frío: create() puede descargar un modelo, así que mostramos progreso en vez de un panel en blanco.
    const onProgress = (fraction: number) => {
      if (!signal.aborted) output.show(`Preparando modelo… ${Math.round(fraction * 100)}%`, "streaming");
    };
    if (!output.text) output.show("Preparando modelo…", "streaming");

    // Guarda el paso actual para que un fallo genérico diga dónde ocurrió (mirar la consola en escena).
    let step = "support check";
    let pair: [string, string] | undefined;
    try {
      // Regla de oro, paso 1: feature detection.
      if (!hasApi("LanguageDetector") || !hasApi("Translator")) {
        throw new Error("Translator / LanguageDetector no soportados (solo Chrome desktop).");
      }

      // 1) ¿En qué idioma escribió el usuario? Sin selector de origen.
      step = "detect";
      const { language, confidence } = await detectLanguage(input, signal, onProgress);
      setDetected(`Idioma detectado: ${language} (${confidence}%)`);

      // Mismo idioma de origen y destino: no hay nada que traducir.
      if (language === targetLanguage) {
        output.show(input);
        return;
      }

      // 2) Traducir con streaming: el texto aparece mientras se genera.
      pair = [language, targetLanguage];
      step = `translator ${language}→${targetLanguage}`;
      const stream = await translateStreaming(input, language, targetLanguage, signal, onProgress);
      step = `stream ${language}→${targetLanguage}`;
      await streamText(stream, output.paint, signal);
      if (!signal.aborted) output.setState("done");
    } catch (error) {
      // Una ejecución superada nunca debe expulsar la instancia de la ejecución más nueva.
      if (signal.aborted || isAbortError(error)) return;
      console.error(`[translate] failed at step: ${step}`);
      // Expulsa lo que falló para que el siguiente intento se cure solo, sin recargar la página.
      if (isRuntimeFailure(error)) {
        if (pair) resetTranslator(...pair);
        else resetDetector();
      }
      output.fail(error);
    }
  }

  // Traducción "en tiempo real": espera una pausa de 400 ms al escribir.
  // La tecla cuenta como activación del usuario, así que create() puede descargar si hace falta.
  function onInput(value: string) {
    setText(value);
    clearTimeout(debounceTimer.current);
    debounceTimer.current = setTimeout(() => void run(value, target), 400);
  }

  // Cambiar el idioma destino traduce de inmediato (sin debounce).
  // Cancela antes el debounce pendiente: su closure guarda el destino anterior y,
  // al dispararse, abortaría esta ejecución y traduciría al idioma viejo.
  // `text` ya tiene lo último escrito, así que no se pierde nada.
  function onTarget(value: string) {
    clearTimeout(debounceTimer.current);
    setTarget(value);
    void run(text, value);
  }

  return (
    <div className="grid gap-6 md:grid-cols-2">
      <div className="flex flex-col gap-3">
        <textarea
          className="field resize-y leading-relaxed"
          rows={6}
          value={text}
          onChange={(e) => onInput(e.target.value)}
        />
        <label className="flex items-center gap-3 text-ink-300">
          Traducir a
          <select className="field w-auto" value={target} onChange={(e) => onTarget(e.target.value)}>
            {TARGETS.map((t) => (
              <option key={t.value} value={t.value}>
                {t.label}
              </option>
            ))}
          </select>
        </label>
      </div>
      <div className="flex flex-col gap-3">
        <p className="meta">{detected}</p>
        <Output text={output.text} state={output.state} />
      </div>
    </div>
  );
}

  // Precalienta Gemini Nano al montar: así el primer "Generar" en escena no paga el create() en frío.
  useEffect(() => {
    void warmUpLocal();
  }, []);

  async function run() {
    setBusy(true);
    output.show("Generando…", "streaming");
    setBadge(null);
    try {
      const t0 = performance.now();
      // Misma interfaz para las dos estrategias: (prompt) => { text, source }.
      const generate = strategy === "firebase" ? generateFirebase : generateManual;
      const result = await generate(prompt);
      const ms = Math.round(performance.now() - t0);
      output.show(result.text);
      // La etiqueta es el momento clave de la demo: quién respondió y cuánto tardó.
      setBadge({
        source: result.source,
        text: `${result.source === "on_device" ? "💻 Local (Gemini Nano)" : "☁️ Nube (Gemini)"} · ${ms} ms`,
      });
    } catch (error) {
      output.fail(error);
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="grid gap-6 md:grid-cols-2">
      <div className="flex flex-col gap-3">
        <textarea
          className="field resize-y leading-relaxed"
          rows={5}
          value={prompt}
          onChange={(e) => setPrompt(e.target.value)}
        />
        <div className="flex flex-wrap items-center gap-2">
          <select
            className="field w-auto"
            value={strategy}
            onChange={(e) => setStrategy(e.target.value as Strategy)}
          >
            <option value="manual">Fallback manual</option>
            <option value="firebase">Firebase AI Logic</option>
          </select>
          <button className="btn-primary" onClick={run} disabled={busy}>
            Generar
          </button>
        </div>
      </div>
      <div className="flex flex-col gap-3">
        <p className={`m-0 min-h-[1.4em] ${badge ? BADGE_COLOR[badge.source] : ""}`}>{badge?.text}</p>
        <Output text={output.text} state={output.state} />
      </div>
    </div>
  );
}

components/output.tsx

"use client";

import { useCallback, useState } from "react";
import { errorMessage } from "@/lib/ai/support";

export type OutputState = "idle" | "streaming" | "done" | "error";

/** State for a demo output panel: text plus a visual state (streaming, error...). */
export function useOutput(initial = "") {
  const [text, setText] = useState(initial);
  const [state, setState] = useState<OutputState>("idle");

  const show = useCallback((value: string, next: OutputState = "done") => {
    setText(value);
    setState(next);
  }, []);

  const fail = useCallback((error: unknown) => {
    // Log the full exception so DevTools shows its name, stack and cause.
    console.error("[demo]", error);
    const name = error instanceof Error && error.name !== "Error" ? `${error.name}: ` : "";
    setText(`⚠️ ${name}${errorMessage(error)}`);
    setState("error");
  }, []);

  /** Paint callback for streamText(): marks the panel as streaming. */
  const paint = useCallback((value: string) => {
    setText(value);
    setState("streaming");
  }, []);

  return { text, state, show, fail, paint, setState };
}

export function Output({
  text,
  state,
  mono,
}: {
  text: string;
  state: OutputState;
  mono?: boolean;
}) {
  return (
    <pre
      className={`out ${mono ? "font-mono text-sm" : ""}`}
      data-state={state}
      aria-live="polite"
    >
      {text}
    </pre>
  );
}
